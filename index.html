function injectAndLoad(htmlText, baseURL) { try { let finalHTML = String(htmlText); // Remove bases antigas finalHTML = finalHTML.replace( /]*>/gi, "" ); // Cria a base para imagens, JS, CSS, sons etc. const baseTag = ``; /* * NÃO injeta código dentro do jogo. * * O jogo original precisa executar exatamente * o JavaScript que veio do capítulo. */ if (/]*>/i.test(finalHTML)) { finalHTML = finalHTML.replace( /]*>/i, match => match + baseTag ); } else { finalHTML = ` ${baseTag} ${finalHTML} `; } // Libera Blob anterior if (frame.dataset.blobUrl) { try { URL.revokeObjectURL( frame.dataset.blobUrl ); } catch (_) {} frame.dataset.blobUrl = ""; } const blob = new Blob( [finalHTML], { type: "text/html" } ); const blobURL = URL.createObjectURL(blob); frame.dataset.blobUrl = blobURL; frame.onload = () => { loader.style.display = "none"; /* * Mostra os controles depois que * o jogo realmente carregou. */ if ( useMobileControls && gameStarted ) { hud.style.display = "block"; settingsBar.style.display = "flex"; } try { frame.contentWindow.focus(); } catch (_) {} }; frame.onerror = () => { loader.style.display = "none"; console.error( "Failed to load chapter iframe." ); }; frame.src = blobURL; } catch (error) { console.error( "injectAndLoad error:", error ); loader.style.display = "none"; alert( "Erro ao carregar o capítulo: " + error.message ); } }
