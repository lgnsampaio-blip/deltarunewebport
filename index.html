function injectAndLoad(htmlText, baseURL) { try { let finalHTML = String(htmlText); // Remove qualquer existente finalHTML = finalHTML.replace( /]*>/gi, "" ); // Adiciona a base correta para carregar // JS, CSS, imagens, sons e outros arquivos. const baseTag = ''; // Coloca a base dentro do if (/]*>/i.test(finalHTML)) { finalHTML = finalHTML.replace( /]*>/i, function (match) { return match + baseTag; } ); } else { finalHTML = '' + '' + '' + '' + baseTag + '' + '' + finalHTML + '' + ''; } // Remove Blob anterior if (frame.dataset.blobUrl) { try { URL.revokeObjectURL( frame.dataset.blobUrl ); } catch (_) {} frame.dataset.blobUrl = ""; } // Cria o HTML do capítulo const blob = new Blob( [finalHTML], { type: "text/html" } ); const blobURL = URL.createObjectURL(blob); frame.dataset.blobUrl = blobURL; frame.onload = function () { loader.style.display = "none"; // Mostra os controles somente // depois que o jogo carregou if ( useMobileControls && gameStarted ) { hud.style.display = "block"; settingsBar.style.display = "flex"; } try { frame.contentWindow.focus(); } catch (_) {} }; frame.onerror = function () { loader.style.display = "none"; console.error( "Failed to load chapter iframe." ); }; frame.src = blobURL; } catch (error) { console.error( "injectAndLoad error:", error ); loader.style.display = "none"; alert( "Erro ao carregar o capítulo: " + error.message ); } }
